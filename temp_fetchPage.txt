async function fetchPage(isRedirect = false, originalPage = null) {
            if (currentCategory) {
                fetchCategory(currentCategory);
                return;
            }
            
            // Subpage Breadcrumbs rendering
            if (currentPage.includes('/')) {
                let parts = currentPage.split('/');
                let breadcrumbHtml = parts.map((part, index) => {
                    let path = parts.slice(0, index + 1).join('/');
                    return `<a href="?page=${encodeURIComponent(path)}" style="color:var(--mc-green); text-decoration:none;">${part}</a>`;
                }).join(' <span style="color:#999; margin:0 5px;">/</span> ');
                titleEl.innerHTML = breadcrumbHtml;
            } else {
                titleEl.textContent = currentPage;
            }

            viewerEl.innerHTML = "로딩 중...";
            
            let doc = await fetchDocumentRaw(currentPage);
            let exists = doc !== null;
            let redirectNotice = isRedirect ? `<div style="font-size: 0.9em; color: gray; margin-bottom: 10px;">↳ <a href="?page=${encodeURIComponent(originalPage)}">${originalPage}</a>에서 넘어옴</div>` : '';

            if (exists) {
                originalContent = doc.content;
                currentFileSha = doc.sha;
                
                // Handle Redirect syntax (#redirect 문서명)
                const redirectMatch = originalContent.match(/^#(?:redirect|넘겨주기)\s+(.+)$/im);
                if (redirectMatch && !isRedirect) {
                    const targetPage = redirectMatch[1].trim();
                    currentPage = targetPage;
                    await fetchPage(true, originalPage || urlParams.get('page') || '대문');
                    return;
                }

                
            let htmlToRender = originalContent;
            if (redirectNotice) {
                htmlToRender = `<div style="color: #666; font-size: 0.9em; margin-bottom: 10px;">↳ ${originalPage}에서 넘어옴</div>\n\n` + htmlToRender;
            }
            await renderPage(htmlToRender);
            
            // Parent/Child rendering
            let relHtml = '<div style="margin-bottom: 20px; font-size: 0.9em; padding: 10px; border-left: 4px solid var(--mc-green); background: rgba(0,0,0,0.03);">';
            let hasRel = false;
            
            // Parent Check
            if (currentPage.includes('/')) {
                let parentPage = currentPage.substring(0, currentPage.lastIndexOf('/'));
                relHtml += `<div><strong>상위 문서:</strong> <a href="?page=${encodeURIComponent(parentPage)}" style="color:var(--mc-green); text-decoration:none;">${parentPage}</a></div>`;
                hasRel = true;
            }
            
            // Child Check
            const allDocs = await fetchAllDocuments();
            const children = allDocs.filter(d => d.startsWith(currentPage + '/'));
            if (children.length > 0) {
                relHtml += `<div style="${hasRel ? 'margin-top: 5px;' : ''}"><strong>하위 문서:</strong> `;
                relHtml += children.map(c => `<a href="?page=${encodeURIComponent(c)}" style="color:var(--mc-green); text-decoration:none;">${c.substring(currentPage.length + 1)}</a>`).join(', ');
                relHtml += `</div>`;
                hasRel = true;
            }
            relHtml += '</div>';
            
            if (hasRel) {
                viewerEl.insertAdjacentHTML('afterbegin', relHtml);
            }

                
                // 5. Re-evaluate scripts
                const scripts = viewerEl.querySelectorAll('script');
                scripts.forEach(oldScript => {
                    const newScript = document.createElement('script');
                    Array.from(oldScript.attributes).forEach(attr => newScript.setAttribute(attr.name, attr.value));
                    newScript.appendChild(document.createTextNode(oldScript.innerHTML));
                    oldScript.parentNode.replaceChild(newScript, oldScript);
                });
                
                if (loggedIn) viewerActions.style.display = 'flex';
                
                // 최근 수정 시각 불러오기
                const lastEditedEl = document.getElementById('last-edited-time');
                if (lastEditedEl) {
                    lastEditedEl.innerHTML = "<i class='fa-solid fa-spinner fa-spin'></i> 수정 시각 확인 중...";
                    const commitApiUrl = `https://api.github.com/repos/${repoOwner}/${repoName}/commits?path=SWIKI/docs/${currentPage}.html&per_page=1`;
                    fetch(commitApiUrl, { headers: ghToken ? { 'Authorization': `token ${ghToken}` } : {} })
                        .then(res => res.ok ? res.json() : [])
                        .then(commits => {
                            if (commits.length > 0) {
                                const date = new Date(commits[0].commit.author.date).toLocaleString();
                                const msgMatch = commits[0].commit.message.match(/\[(.*?)\]/);
                                const author = msgMatch ? msgMatch[1] : commits[0].commit.author.name;
                                lastEditedEl.innerHTML = `<i class='fa-regular fa-clock'></i> 최근 수정: ${date} (by <strong>${author}</strong>)`;
                            } else {
                                lastEditedEl.innerHTML = "";
                            }
                        })
                        .catch(() => { lastEditedEl.innerHTML = ""; });
                }
            } else {
                viewerEl.innerHTML = `<div style="text-align:center; padding: 50px 0;">
                    <h2><i class="fa-solid fa-triangle-exclamation"></i> 문서가 존재하지 않습니다</h2>
                    <p style="margin: 15px 0;">'${currentPage}' 문서는 아직 작성되지 않았습니다.</p>
                    ${loggedIn ? `<button class="mc-button primary" onclick="startEdit()">새 문서 작성하기</button>` : `<p style="color:red">문서를 작성하려면 관리자 로그인이 필요합니다.</p>`}
                </div>`;
                const lastEditedEl = document.getElementById('last-edited-time');
                if (lastEditedEl) lastEditedEl.innerHTML = "";
            }
        }

        async function expandTemplates(text, depth = 0) {
            if (depth > 3) return text; // Prevent infinite loops
            
            const templateRegex = /\[include\((.*?)\)\]/g;
            let matches;
            const templatesToFetch = new Set();
            while ((matches = templateRegex.exec(text)) !== null) {
                templatesToFetch.add(matches[1].trim());
            }
            
            if (templatesToFetch.size === 0) return text;
            
            const templateContents = {};
            const promises = Array.from(templatesToFetch).map(async (tpl) => {
                let doc = await fetchDocumentRaw(tpl);
                if (doc) {
                    templateContents[tpl] = await expandTemplates(doc.content, depth + 1);
                } else {
                    templateContents[tpl] = `> **[오류]** '${tpl}' 틀을 불러올 수 없습니다.`;
                }
            });
            
            await Promise.all(promises);
            
            return text.replace(templateRegex, (match, p1) => {
                return templateContents[p1.trim()];
            });
        }

        function parseMarkdown(text) {
            let codeBlocks = [];
            text = text.replace(/```([\s\S]*?)```/g, (match, p1) => {
                codeBlocks.push(`<pre><code>${p1.replace(/</g, '&lt;').replace(/>/g, '&gt;')}</code></pre>`);
                return `%%%CODEBLOCK_${codeBlocks.length - 1}%%%`;
            });
            text = text.replace(/`([^`]+)`/g, (match, p1) => {
                codeBlocks.push(`<code>${p1.replace(/</g, '&lt;').replace(/>/g, '&gt;')}</code>`);
                return `%%%CODEBLOCK_${codeBlocks.length - 1}%%%`;
            });

            text = text.replace(/^(#{1,6})\s+(.*)$/gm, (match, hashes, content) => {
                let level = hashes.length;
                return `<h${level}>${content}</h${level}>`;
            });
            text = text.replace(/^>\s+(.*)$/gm, '<blockquote>$1</blockquote>');
            text = text.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
            text = text.replace(/__(.*?)__/g, '<strong>$1</strong>');
            text = text.replace(/\*(.*?)\*/g, '<em>$1</em>');
            text = text.replace(/_(.*?)_/g, '<em>$1</em>');
            text = text.replace(/~~(.*?)~~/g, '<del>$1</del>');
            text = text.replace(/!\[(.*?)\]\((.*?)\)/g, '<img src="$2" alt="$1" style="max-width:100%;">');
            text = text.replace(/\[(.*?)\]\((.*?)\)/g, '<a href="$2" target="_blank">$1</a>');

            let inList = false;
            let lines = text.split('\n');
            for (let i = 0; i < lines.length; i++) {
                if (lines[i].match(/^\s*[-*]\s+(.*)$/)) {
                    lines[i] = lines[i].replace(/^\s*[-*]\s+(.*)$/, '<li>$1</li>');
                    if (!inList) { lines[i] = '<ul>' + lines[i]; inList = true; }
                } else if (inList) {
                    lines[i - 1] += '</ul>';
                    inList = false;
                }
            }
            if (inList) lines[lines.length - 1] += '</ul>';
            text = lines.join('\n');

            let inTable = false;
            lines = text.split('\n');
            for (let i = 0; i < lines.length; i++) {
                let line = lines[i].trim();
                if (line.startsWith('|') && line.endsWith('|')) {
                    if (line.replace(/\s/g, '').match(/^\|[-:|]+\|$/)) {
                        lines[i] = ''; 
                        continue;
                    }
                    let cells = line.substring(1, line.length - 1).split('|').map(c => c.trim());
                    let rowHtml = '<tr>' + cells.map(c => `<td>${c}</td>`).join('') + '</tr>';
                    
                    if (!inTable) {
                        lines[i] = '<table class="mc-table" style="border-collapse:collapse; width:auto; border:2px solid var(--mc-stone); margin-bottom: 15px;"><tbody>' + rowHtml;
                        inTable = true;
                    } else {
                        lines[i] = rowHtml;
                    }
                } else if (inTable) {
                    lines[i - 1] += '</tbody></table>';
                    inTable = false;
                }
            }
            if (inTable) lines[lines.length - 1] += '</tbody></table>';
            text = lines.join('\n');

            text = text.replace(/\n\n+/g, '<br><br>');
            text = text.replace(/([^>])\n([^<])/g, '$1<br>$2');

            codeBlocks.forEach((code, index) => {
                text = text.replace(`%%%CODEBLOCK_${index}%%%`, code);
            });

            return text;
        }

        async function renderPage(content) {
            try {
                content = await expandTemplates(content);
                
                let currentCategories = [];
                content = content.replace(/\[\[분류:(.*?)\]\]/g, (match, p1) => {
                    currentCategories.push(p1.trim());
                    return "";
                });
                
                let html = parseMarkdown(content);
                
                if (currentCategories.length > 0) {
                    let catHtml = '<div style="margin-top: 30px; padding: 10px; border: 1px solid var(--mc-stone); border-radius: 5px; background: rgba(0,0,0,0.05);">';
                    catHtml += '<strong>분류: </strong>';
                    catHtml += currentCategories.map(c => `<a href="?page=${encodeURIComponent(c)}" style="color:var(--mc-green); text-decoration:none;">${c}</a>`).join(' | ');
                    catHtml += '</div>';
                    html += catHtml;
                }
                
                viewerEl.innerHTML = html;\n                if (typeof generateTOC === \'function\') generateTOC(viewerEl);\n                if (typeof renderFootnotes === \'function\') renderFootnotes(viewerEl);
            } catch(e) {
                viewerEl.innerHTML = `<div style="color:red; font-weight:bold;">렌더링 중 오류가 발생했습니다: ${e.message}</div>`;
                console.error(e);
            }
        }
